param name string
param location string
param addressPrefix string
param subnets array = []

resource vnet 'Microsoft.Network/virtualNetworks@2023-09-01' = {
  name: name
  location: location
  properties: {
    addressSpace: {
      addressPrefixes: [ addressPrefix ]
    }
    subnets: subnets
  }
}

output id string = vnet.id
output name string = vnet.name
